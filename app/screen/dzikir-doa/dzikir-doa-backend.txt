// Ganti bagian data dummy dengan ini:
const [dzikirData, setDzikirData] = useState<DzikirDoaItem[]>([]);
const [loading, setLoading] = useState(true);

useEffect(() => {
  fetchDzikirDoa();
}, []);

const fetchDzikirDoa = async () => {
  try {
    setLoading(true);
    const response = await axios.get(`${API_BASE_URL}/dzikir-doa`);
    setDzikirData(response.data);
  } catch (error) {
    console.error("Failed to fetch dzikir doa:", error);
  } finally {
    setLoading(false);
  }
};
